% #1=节点名前缀 #2=yshift #3=基因/颜色列表 → 生成 #11,#12,...,#19
\newcommand{\GeneRowPref}[3]{%
  \begin{scope}[yshift=#2]
    \foreach \g/\c [count=\i] in {#3} {
      \node[gene, fill=\c] (#1\i) at ({(\i-1)*\genestep},0) {\g};
    }
  \end{scope}
}
